cmake_minimum_required(VERSION 3.20)
project(hush VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(HUSH_TESTS "Build tests" ON)
option(HUSH_SANITIZE "Build with address + undefined-behavior sanitizers" OFF)

add_library(hush STATIC src/engine.cpp src/rules_detector.cpp src/c_api.cpp)
target_include_directories(hush PUBLIC include)
if(MSVC)
  target_compile_options(hush PRIVATE /W4)
else()
  target_compile_options(hush PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wno-sign-conversion)
endif()
if(HUSH_SANITIZE)
  target_compile_options(hush PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(hush PUBLIC -fsanitize=address,undefined)
  target_compile_definitions(hush PUBLIC HUSH_SANITIZE)
endif()

option(HUSH_TOOLS "Build the hush_run command-line tool" ON)
if(HUSH_TOOLS)
  add_executable(hush_run tools/hush_run.cpp)
  target_link_libraries(hush_run PRIVATE hush)
endif()

if(HUSH_TESTS)
  enable_testing()
  add_executable(hush_tests tests/test_engine.cpp)
  target_link_libraries(hush_tests PRIVATE hush)
  add_test(NAME hush_tests COMMAND hush_tests)
endif()
